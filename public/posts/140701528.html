class Solution {
    public int singleNonDuplicate(int[] nums) {
        int left = 0;
        int right = nums.length - 1;
        // 常规情况，重复元素偶数位第一次出现，奇数位第二次出现，如果相反，说明前置被重复元素插入打乱顺序了
        while (left < right) {
            int mid = left + (right - left) / 2;
            // 下标是偶数，如果未
            if (mid % 2 == 0) {
                if (nums[mid] == nums[mid + 1]) {
                    // 目标值在右侧
                    left = mid + 2;
                } else {
                    // 无法进一步判断，mid自身有可能就是目标值
                    right = mid;
                }
            } else {
                if (nums[mid] == nums[mid - 1]) {
                    // 目标值在右侧
                    left = mid + 1;
                } else {
                    // 无法进一步判断，mid自身有可能就是目标值
                    right = mid;
                }

            }
        }
        return nums[left];
    }
}
